country_event = {
	id = krcw_sph.1001
	title = krcw_sph_1001_t
	desc = krcw_sph_1001_d
	picture = GFX_report_event_generic_conference
	is_triggered_only = yes
	trigger = {
		country_exists = SOV country_exists = ALB
	}
	option = {
		name = krcw_sph_rival.a
		trigger = { has_political_power > 49 }
		ai_chance = {
			base = 40
			modifier = { factor = 0 has_political_power < 80 }
		}
		add_political_power = -50
		hidden_effect = {
			ALB = {
				add_to_variable = { krcw_inf_ITA = 10 } clamp_variable = { var = krcw_inf_ITA min = 0 max = 100 }
				add_to_variable = { krcw_inf_SOV = -10 } clamp_variable = { var = krcw_inf_SOV min = 0 max = 100 }
				set_country_flag = { flag = krcw_sph_outbid days = 30 value = 1 }
			}
		}
		custom_effect_tooltip = krcw_sph_tt_outbid
	}
	option = {
		name = krcw_sph_rival.b
		ai_chance = {
			base = 30
		}
		krcw_tension_up_small = yes
		hidden_effect = { ALB = { set_country_flag = { flag = krcw_sph_warned days = 30 value = 1 } } }
		custom_effect_tooltip = krcw_sph_tt_warned
		if = { limit = { country_exists = SOV } add_opinion_modifier = { target = SOV modifier = krcw_sph_rivalry } SOV = { add_opinion_modifier = { target = ROOT modifier = krcw_sph_rivalry } } }
	}
	option = {
		name = krcw_sph_rival.c
		ai_chance = {
			base = 30
		}
	}
}
