country_event = {
	id = krcw_proxy.1
	title = krcw_proxy_1_t
	desc = krcw_proxy_1_d
	picture = GFX_report_event_generic_rally
	trigger = {
		tag = GRE
		has_war = no
		has_global_flag = krcw_proxy_gre_stage2
		NOT = { has_global_flag = krcw_proxy_gre_done }
	}
	mean_time_to_happen = { days = 20 }
	fire_only_once = yes
	option = {
		name = krcw_proxy_1_a
		ai_chance = { base = 100 }
		set_global_flag = krcw_proxy_gre_done
		set_global_flag = krcw_proxy_gre
		add_to_variable = { global.krcw_proxy_active = 1 }
		set_global_flag = { flag = krcw_proxy_cooldown days = 300 value = 1 }
		add_popularity = { ideology = trotskyism popularity = 0.15 }
		start_civil_war = {
			ideology = trotskyism
			# 0.35 rebels were crushed before any bloc could intervene; KR uses 0.5 for its Greek rising
			size = 0.45
			# no fleet: a rebel navy let the losing side hold out on islands and the war never ended
			navy_ratio = 0
			# a fixed, contiguous mainland block the rebels' patron can reach (proxy_prelude.REBELS)
			capital = 185
			states = { 185 977 790 }
			save_global_event_target_as = krcw_proxy_gre_rebels
			# CWIC sets CIVIL_WAR_INVOLVEMENT_MIN_TENSION = 0 (IC_Defines.lua), so the engine may pull
			# other powers into any civil war, and the revolter could be offered a faction seat. Keep the
			# rebels out of every faction so the war stays a proxy war (and can receive volunteers).
			set_rule = { can_join_factions = no }
			# the rebels' own name and flag: the revolter copies the host's cosmetic tag otherwise, and
			# both sides of the Persian war were called the Imperial State of Iran
			set_cosmetic_tag = KRCW_GRE_REB
		}
		# two years before the war is settled one way or the other (krcw_proxy.7xx)
		set_global_flag = { flag = krcw_proxy_gre_clock days = 730 value = 1 }
		# a year of guerrilla resilience so the war lasts long enough for outside help to matter
		# (TFR gives a losing civil-war side a defence buff for the same reason)
		event_target:krcw_proxy_gre_rebels = { add_timed_idea = { idea = krcw_proxy_insurgency days = 365 } }
		# a belligerent movement in the ceasefire regime (patch_un.py / geneva.py), not an ordinary state
		event_target:krcw_proxy_gre_rebels = { add_ideas = unrecognized_state }
		# the rebels' own leader and portrait (proxy_prelude.LEADERS); without one the engine invents a nameless leader
		event_target:krcw_proxy_gre_rebels = {
			set_politics = { ruling_party = trotskyism elections_allowed = no }
			if = { limit = { NOT = { has_character = GRE_krcw_velouchiotis } } recruit_character = GRE_krcw_velouchiotis }
			promote_character = GRE_krcw_velouchiotis
		}
		# each side asks its patron for troops
		if = { limit = { country_exists = SOV } SOV = { country_event = { id = krcw_proxy.313 days = 10 } } }
		if = { limit = { country_exists = FRA } FRA = { country_event = { id = krcw_proxy.314 days = 20 } } }
		krcw_tension_up_small = yes
		custom_effect_tooltip = krcw_proxy_tt
		random_country = { news_event = { id = krcw_proxy.101 days = 1 } }
	}
	option = {
		name = krcw_proxy_1_b
		ai_chance = { base = 0 }
		set_global_flag = krcw_proxy_gre_done
		add_political_power = -100
		add_stability = -0.05
	}
}
