			custom_effect_tooltip = krcw_sov_prep_paid_tt

		else = {
			add_to_variable = { krcw_sov_ext = 8 }
			add_to_variable = { krcw_sov_mon = 5 }
			add_stability = -0.05
			custom_effect_tooltip = krcw_sov_prep_missed_tt
		}
		clamp_variable = { var = krcw_sov_mil min = 0 max = 100 }
		clamp_variable = { var = krcw_sov_ext min = 0 max = 100 }
		clamp_variable = { var = krcw_sov_mon min = 0 max = 100 }
		# an interim Supreme Ruler every faction can tolerate
		recruit_character = SOV_krcw_nekrasov
		promote_character = SOV_krcw_nekrasov
		krcw_tension_up_medium = yes
		random_country = { news_event = { id = krcw_sov.3 days = 1 } }
		# the Council deadlock opens the struggle
		country_event = { id = krcw_sov.4 days = 20 random_days = 10 }
		# the Baltic/Finnish question resurfaces during the interregnum (~1956)
		country_event = { id = krcw_sov.9 days = 315 random_days = 120 }

		ai_chance = {
			base = 1
		}
