country_event = {
	id = krcw_tibrev.6
	hidden = yes
	trigger = {
		tag = TIB
		has_global_flag = KRCW_tibet_revolt_started
		NOT = { has_global_flag = KRCW_tibet_revolt_over }
	}
	is_triggered_only = yes
	option = {
		ai_chance = {
			base = 1
		}
		if = {
			limit = { has_war_with = CHI }
			set_global_flag = KRCW_tibet_war_began
			log = "[GetDateText] KRCWTIB war on (retry=[?global.krcw_tib_retry|0])"
		}
		else_if = {
			limit = { NOT = { check_variable = { global.krcw_tib_retry > 0 } } }
			set_variable = { global.krcw_tib_retry = 1 }
			log = "[GetDateText] KRCWTIB declare failed (faction=[?is_in_faction] subject=[?is_subject]), CHI retries"
			if = { limit = { is_subject_of = CHI } CHI = { end_puppet = TIB } }
			if = { limit = { is_in_faction = yes } leave_faction = yes }
			CHI = { set_country_flag = { flag = krcw_limited_war value = 1 days = 2 } }
			CHI = { if = { limit = { NOT = { has_war_with = TIB } } declare_war_on = { target = TIB type = annex_everything } } }
			country_event = { id = krcw_tibrev.6 days = 1 }
		}
		else_if = {
			limit = { check_variable = { global.krcw_tib_retry < 2 } }
			set_variable = { global.krcw_tib_retry = 2 }
			log = "[GetDateText] KRCWTIB CHI still cannot declare, TIB declares a war of independence"
			if = { limit = { is_in_faction = yes } leave_faction = yes }
			set_country_flag = { flag = krcw_limited_war value = 1 days = 2 }
			declare_war_on = { target = CHI type = independence_wargoal }
			country_event = { id = krcw_tibrev.6 days = 1 }
		}
		else = {
			log = "[GetDateText] KRCWTIB no war possible, status quo restored"
			CHI = {
				if = { limit = { has_dlc = "Together for Victory" }
					set_autonomy = { target = TIB autonomous_state = autonomy_China_Local_Government freedom_level = 0.9 }
				}
				else = { puppet = TIB }
			}
			set_global_flag = KRCW_tibet_revolt_over
			set_global_flag = KRCW_tibet_outcome_suppressed
			CHI = { remove_ideas = krcw_chi_tibet_pacification remove_ideas = krcw_chi_frontier_garrisons }
			TIB = { remove_ideas = krcw_tib_protectors_of_faith remove_ideas = krcw_tib_mountain_war }
			if = { limit = { country_exists = TIB TIB = { is_subject = no is_in_faction_with = CHI } } TIB = { leave_faction = yes } }
			RAJ = { remove_ideas = krcw_raj_arming_tibet }
			news_event = { id = krcw_tibrev.30 days = 1 }
		}
	}
}
