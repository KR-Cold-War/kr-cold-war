# KRCW world background window for every human player at game start (build/world_intro.py)
on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = { is_ai = no NOT = { has_country_flag = krcw_world_intro_seen } }
				set_country_flag = krcw_world_intro_seen
				set_country_flag = krcw_world_intro_open
			}
		}
	}
}
