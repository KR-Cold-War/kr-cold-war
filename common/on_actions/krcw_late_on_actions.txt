		effect = {
			# Act 3 schedules (1966-78)
			FRA = { country_event = { id = krcw_late.1 days = 1669 random_days = 1 } }
			CHI = { country_event = { id = krcw_late.10 days = 1840 random_days = 1 } }
			# Dark branches
			SOV = { country_event = { id = krcw_late.24 days = 1776 random_days = 3 } }

			# Dark branches Act 4

		}
	}
	# Second Commune response chain
	on_monthly_FRA = {
		effect = {
			if = {
				limit = {
					has_country_flag = KRCW_fra_68_happened
					has_country_flag = KRCW_fra_68_reform
					NOT = { has_global_flag = KRCW_fra_68_charter }
					date > 1954.2.4
				}
				country_event = { id = krcw_late.2 days = 60 }
			}
			if = {
				limit = {
					has_country_flag = KRCW_fra_68_happened
					has_country_flag = KRCW_fra_68_repression
					NOT = { has_global_flag = KRCW_fra_68_technocracy }
					date > 1954.2.4
				}
				country_event = { id = krcw_late.3 days = 60 }
			}
			if = {
				limit = {
					has_country_flag = KRCW_fra_68_happened
					has_country_flag = KRCW_fra_68_radicalisation
					NOT = { has_global_flag = KRCW_fra_68_split }
					date > 1954.2.11
				}
				country_event = { id = krcw_late.4 days = 90 }
			}
		}
	}
}
