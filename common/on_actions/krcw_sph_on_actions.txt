	on_monthly_ALB = {
		effect = {
			if = {
				limit = { has_global_flag = KRCW_active }
				if = {
					limit = { country_exists = SOV has_government = SOV check_variable = { krcw_inf_SOV < 50 } }
					add_to_variable = { krcw_inf_SOV = 1 } clamp_variable = { var = krcw_inf_SOV min = 0 max = 100 }
				}
				else_if = {
					limit = {
						check_variable = { krcw_inf_SOV > 0 }
						NOT = { has_country_flag = krcw_sph_treaty_SOV has_country_flag = krcw_sph_aligned_SOV }
						NOT = { AND = { country_exists = SOV is_subject_of = SOV } }
					}
					add_to_variable = { krcw_inf_SOV = -1 } clamp_variable = { var = krcw_inf_SOV min = 0 max = 100 }
				}
				if = {
