on_actions = {
	on_startup = {
		effect = {
			set_variable = { global.krcw_ti_crisis = 10 }
			set_variable = { global.krcw_ma_crisis = 15 }
		}
	}
	on_monthly_FRA = {
		effect = {
			if = {
				limit = { has_global_flag = KRCW_active NOT = { has_global_flag = KRCW_world_war } NOT = { has_global_flag = KRCW_cold_war_over } is_faction_leader = yes }
				if = { limit = { has_stability < 0.45 } add_to_variable = { global.krcw_ti_crisis = 1 } clamp_variable = { var = global.krcw_ti_crisis min = 0 max = 100 } }
				if = { limit = { has_stability < 0.30 } add_to_variable = { global.krcw_ti_crisis = 1 } clamp_variable = { var = global.krcw_ti_crisis min = 0 max = 100 } }
				if = { limit = { has_political_power < 0 } add_to_variable = { global.krcw_ti_crisis = 1 } clamp_variable = { var = global.krcw_ti_crisis min = 0 max = 100 } }
				if = { limit = { has_war = yes has_war_support < 0.40 } add_to_variable = { global.krcw_ti_crisis = 1 } clamp_variable = { var = global.krcw_ti_crisis min = 0 max = 100 } }
				if = { limit = { has_stability > 0.60 } add_to_variable = { global.krcw_ti_crisis = -2 } clamp_variable = { var = global.krcw_ti_crisis min = 0 max = 100 } }
				# the bloc economy (build/economy.py): a leader in crisis, and a bloc full of distressed members
				if = { limit = { has_variable = krcw_econ_tier check_variable = { krcw_econ_tier < 2 } } add_to_variable = { global.krcw_ti_crisis = 1 } clamp_variable = { var = global.krcw_ti_crisis min = 0 max = 100 } }
				if = { limit = { has_variable = krcw_econ_tier check_variable = { krcw_econ_tier < 1 } } add_to_variable = { global.krcw_ti_crisis = 1 } clamp_variable = { var = global.krcw_ti_crisis min = 0 max = 100 } }
				if = { limit = { has_variable = krcw_econ_tier check_variable = { krcw_econ_tier > 4 } } add_to_variable = { global.krcw_ti_crisis = -1 } clamp_variable = { var = global.krcw_ti_crisis min = 0 max = 100 } }
				if = { limit = { check_variable = { global.krcw_econ_ti_distress > 2 } } add_to_variable = { global.krcw_ti_crisis = 1 } clamp_variable = { var = global.krcw_ti_crisis min = 0 max = 100 } }
				if = {
					limit = { check_variable = { global.krcw_ti_crisis > 39 } NOT = { has_global_flag = krcw_col_ti_1 } }
					set_global_flag = krcw_col_ti_1
					country_event = { id = krcw_col.4 days = 1 }
				}
				if = {
					limit = { check_variable = { global.krcw_ti_crisis > 69 } NOT = { has_global_flag = krcw_col_ti_2 } }
					set_global_flag = krcw_col_ti_2
					random_country = {
						limit = { is_in_faction_with = ROOT NOT = { tag = ROOT } is_ai = yes }
						leave_faction = yes
					}
					country_event = { id = krcw_col.5 days = 1 }
				}
				if = {
					limit = { check_variable = { global.krcw_ti_crisis > 99 } NOT = { has_global_flag = krcw_col_ti_3 } }
					set_global_flag = krcw_col_ti_3
					country_event = { id = krcw_col.6 days = 1 }
				}
			}
		}
	}
	on_monthly_SOV = {
		effect = {
			if = {
				limit = { has_global_flag = KRCW_active NOT = { has_global_flag = KRCW_world_war } NOT = { has_global_flag = KRCW_cold_war_over } is_faction_leader = yes }

				if = { limit = { has_stability < 0.30 } add_to_variable = { global.krcw_ma_crisis = 1 } clamp_variable = { var = global.krcw_ma_crisis min = 0 max = 100 } }
				if = { limit = { has_political_power < 0 } add_to_variable = { global.krcw_ma_crisis = 1 } clamp_variable = { var = global.krcw_ma_crisis min = 0 max = 100 } }
				if = { limit = { has_war = yes has_war_support < 0.40 } add_to_variable = { global.krcw_ma_crisis = 1 } clamp_variable = { var = global.krcw_ma_crisis min = 0 max = 100 } }
				if = { limit = { has_stability > 0.60 } add_to_variable = { global.krcw_ma_crisis = -2 } clamp_variable = { var = global.krcw_ma_crisis min = 0 max = 100 } }
				# the bloc economy (build/economy.py): a leader in crisis, and a bloc full of distressed members
				if = { limit = { has_variable = krcw_econ_tier check_variable = { krcw_econ_tier < 2 } } add_to_variable = { global.krcw_ma_crisis = 1 } clamp_variable = { var = global.krcw_ma_crisis min = 0 max = 100 } }
				if = { limit = { has_variable = krcw_econ_tier check_variable = { krcw_econ_tier < 1 } } add_to_variable = { global.krcw_ma_crisis = 1 } clamp_variable = { var = global.krcw_ma_crisis min = 0 max = 100 } }
				if = { limit = { has_variable = krcw_econ_tier check_variable = { krcw_econ_tier > 4 } } add_to_variable = { global.krcw_ma_crisis = -1 } clamp_variable = { var = global.krcw_ma_crisis min = 0 max = 100 } }
				if = { limit = { check_variable = { global.krcw_econ_ma_distress > 2 } } add_to_variable = { global.krcw_ma_crisis = 1 } clamp_variable = { var = global.krcw_ma_crisis min = 0 max = 100 } }
				if = {
					limit = { check_variable = { global.krcw_ma_crisis > 39 } NOT = { has_global_flag = krcw_col_ma_1 } }
					set_global_flag = krcw_col_ma_1
					country_event = { id = krcw_col.1 days = 1 }
				}
				if = {
					limit = { check_variable = { global.krcw_ma_crisis > 69 } NOT = { has_global_flag = krcw_col_ma_2 } }
					set_global_flag = krcw_col_ma_2

					}
					country_event = { id = krcw_col.2 days = 1 }
				}
				if = {
					limit = { check_variable = { global.krcw_ma_crisis > 99 } NOT = { has_global_flag = krcw_col_ma_3 } }
					set_global_flag = krcw_col_ma_3

	}
