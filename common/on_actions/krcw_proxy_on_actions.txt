		effect = {
			set_variable = { global.krcw_proxy_active = 0 }
		}
	}
	on_monthly_FRA = {
		effect = {
		if = {
			limit = { has_global_flag = krcw_proxy_gre OR = { NOT = { country_exists = GRE } GRE = { has_war = no } } }

			add_to_variable = { global.krcw_proxy_active = -1 }
			clamp_variable = { var = global.krcw_proxy_active min = 0 max = 5 }

	on_weekly_BRM = {
		effect = {
			if = {
				limit = { has_global_flag = krcw_proxy_brm has_event_target = krcw_proxy_brm_rebels NOT = { has_global_flag = krcw_proxy_brm_ending } }
				if = {
					limit = { NOT = { has_global_flag = krcw_proxy_brm_started } }
					set_global_flag = krcw_proxy_brm_started
					set_global_flag = { flag = krcw_proxy_brm_grace days = 60 value = 1 }
					event_target:krcw_proxy_brm_rebels = { set_variable = { global.krcw_proxy_brm_reb0 = num_of_controlled_states } set_variable = { global.krcw_proxy_brm_rsp = surrender_progress } }
					set_variable = { global.krcw_proxy_brm_gov0 = num_of_controlled_states }
					log = "[GetDateText] KRCWPROXY BRM start rebel_states=[?global.krcw_proxy_brm_reb0|0] gov_states=[?global.krcw_proxy_brm_gov0|0] gov_sp=[?surrender_progress|2] rebel_sp=[?global.krcw_proxy_brm_rsp|2]"
				}
				else = {
					# gain: states the rebels hold beyond the ones they rose with (negative once they lose ground)
					event_target:krcw_proxy_brm_rebels = { set_variable = { global.krcw_proxy_brm_reb = num_of_controlled_states } }
					set_variable = { global.krcw_proxy_brm_gain = global.krcw_proxy_brm_reb }
					subtract_from_variable = { global.krcw_proxy_brm_gain = global.krcw_proxy_brm_reb0 }
					set_variable = { global.krcw_proxy_brm_half = global.krcw_proxy_brm_reb }
					multiply_variable = { global.krcw_proxy_brm_half = 2 }
					# escalation: asked at 0.2 (was 0.3), and the allies get 90 days to arrive. In the 2026-10-08 test Greece fell
					# at 0.67 eight weeks after the rising; between the request at 0.3 and the fall at 0.6 the allies had no time.
					if = {
						limit = { NOT = { has_global_flag = krcw_proxy_brm_escalated } OR = { surrender_progress > 0.2 check_variable = { global.krcw_proxy_brm_gain > 1 } } }
						set_global_flag = krcw_proxy_brm_escalated
						set_global_flag = { flag = krcw_proxy_brm_relief days = 90 value = 1 }
						if = { limit = { country_exists = CHI } CHI = { country_event = { id = krcw_proxy.641 days = 3 } } }
					}
					if = {
						limit = {
							OR = {
								# the engine already ended the war (the rebels capitulated and were annexed)
								event_target:krcw_proxy_brm_rebels = { exists = no }
								# crushed: past the grace period the rebels hold less than half the states they rose with
								AND = {
									NOT = { has_global_flag = krcw_proxy_brm_grace }
									check_variable = { global.krcw_proxy_brm_half < global.krcw_proxy_brm_reb0 }
								}
								# two years on: the government wins unless the rebels have gained ground or it is losing
								AND = {
									NOT = { has_global_flag = krcw_proxy_brm_clock }
									check_variable = { global.krcw_proxy_brm_gain < 2 }
									NOT = { surrender_progress > 0.3 }
								}
							}
						}
						set_global_flag = krcw_proxy_brm_ending
						log = "[GetDateText] KRCWPROXY BRM crushed rebel_states=[?global.krcw_proxy_brm_reb|0]/[?global.krcw_proxy_brm_reb0|0] gov_sp=[?surrender_progress|2]"
						country_event = { id = krcw_proxy.741 }
					}
					else_if = {
						limit = {
							OR = {
								288 = { is_controlled_by = event_target:krcw_proxy_brm_rebels }
								# while the allies are on their way only near-collapse ends it (the engine settles a civil war at 1.0)
								surrender_progress > 0.8
								AND = { surrender_progress > 0.6 NOT = { has_global_flag = krcw_proxy_brm_relief } }
								NOT = { has_global_flag = krcw_proxy_brm_clock }
							}
						}
						set_global_flag = krcw_proxy_brm_ending
						log = "[GetDateText] KRCWPROXY BRM falls rebel_states=[?global.krcw_proxy_brm_reb|0]/[?global.krcw_proxy_brm_reb0|0] gov_sp=[?surrender_progress|2]"
						country_event = { id = krcw_proxy.742 }
					}
				}
			}
		}
