krcw_ww_begin_from_FRA = {
	if = {
		limit = { NOT = { has_global_flag = KRCW_world_war } FRA = { exists = yes } SOV = { exists = yes } }
		FRA = { declare_war_on = { target = SOV type = annex_everything } }
		every_country = {
			limit = { is_in_faction_with = FRA NOT = { tag = FRA } NOT = { has_war_with = SOV } }
			add_to_war = { targeted_alliance = FRA enemy = SOV hostility_reason = asked_to_join }
		}
		every_country = {
			limit = { is_in_faction_with = SOV NOT = { tag = SOV } NOT = { has_war_with = FRA } }
			add_to_war = { targeted_alliance = SOV enemy = FRA hostility_reason = asked_to_join }
		}
		set_global_flag = KRCW_world_war
		news_event = { id = krcw_ww.3 days = 1 }
	}
}
