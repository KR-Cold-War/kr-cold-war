krcw_restart_american_war = {
	set_global_flag = KRCW_american_war_restarted
	set_variable = { global.krcw_ohio_tension = 100 }
	if = { limit = { country_exists = WCA WCA = { is_in_faction = yes } } set_global_flag = KRCW_wca_was_in_faction WCA = { leave_faction = yes } }
	if = { limit = { country_exists = USA USA = { is_in_faction = yes } } set_global_flag = KRCW_usa_was_in_faction USA = { leave_faction = yes } }
	every_country = {
		limit = { OR = { tag = USA tag = WCA } }
		if = { limit = { has_idea = krcw_mississippi_line } remove_ideas = krcw_mississippi_line }
		if = { limit = { has_idea = krcw_ohio_line } remove_ideas = krcw_ohio_line }
		add_ideas = krcw_second_civil_war_phase
	}
	if = { limit = { tag = USA } declare_war_on = { target = WCA type = annex_everything } WCA = { country_event = { id = krcw_ohio.52 days = 1 } } }
	if = { limit = { tag = WCA } declare_war_on = { target = USA type = annex_everything } USA = { country_event = { id = krcw_ohio.52 days = 1 } } }
	krcw_tension_up_large = yes
	krcw_detente_down_medium = yes
	random_country = { news_event = { id = krcw_ohio.53 days = 2 } }
	if = { limit = { country_exists = FRA } FRA = { country_event = { id = krcw_ohio.54 days = 5 } } }
	if = { limit = { country_exists = SOV } SOV = { country_event = { id = krcw_ohio.55 days = 5 } } }
}

krcw_end_american_war = {
	set_global_flag = KRCW_american_war_over
	clr_global_flag = KRCW_american_war_restarted
	every_country = {
		limit = { OR = { tag = USA tag = WCA } }
		if = { limit = { has_idea = krcw_second_civil_war_phase } remove_ideas = krcw_second_civil_war_phase }
	}
	if = { limit = { has_global_flag = KRCW_wca_was_in_faction country_exists = WCA country_exists = FRA WCA = { is_in_faction = no } } FRA = { add_to_faction = WCA } }
	if = { limit = { has_global_flag = KRCW_usa_was_in_faction country_exists = USA country_exists = SOV USA = { is_in_faction = no } } SOV = { add_to_faction = USA } }
	clr_global_flag = KRCW_wca_was_in_faction
	clr_global_flag = KRCW_usa_was_in_faction
}
