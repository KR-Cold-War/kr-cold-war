# population, education and the two auto-picked project targets, in one pass over the owned states
krcw_dev_census = {
	set_temp_variable = { krcw_t_pop = 0 }
	set_temp_variable = { krcw_ts_literate_educated = 0 }
	set_temp_variable = { krcw_ts_primary_educated = 0 }
	set_temp_variable = { krcw_ts_high_school_educated = 0 }
	set_temp_variable = { krcw_ts_college_educated = 0 }
	set_temp_variable = { krcw_t_best = 0 }
	set_temp_variable = { krcw_t_rail = 0 }
	clear_variable = krcw_dev_target
	clear_variable = krcw_dev_rail_target
	every_owned_state = {
		set_temp_variable = { krcw_tp = state_population_k }
		divide_temp_variable = { krcw_tp = 1000 }
		add_to_temp_variable = { krcw_t_pop = krcw_tp }
		if = { limit = { has_variable = literate_educated } set_temp_variable = { krcw_tx = literate_educated } multiply_temp_variable = { krcw_tx = krcw_tp } add_to_temp_variable = { krcw_ts_literate_educated = krcw_tx } }
		if = { limit = { has_variable = primary_educated } set_temp_variable = { krcw_tx = primary_educated } multiply_temp_variable = { krcw_tx = krcw_tp } add_to_temp_variable = { krcw_ts_primary_educated = krcw_tx } }
		if = { limit = { has_variable = high_school_educated } set_temp_variable = { krcw_tx = high_school_educated } multiply_temp_variable = { krcw_tx = krcw_tp } add_to_temp_variable = { krcw_ts_high_school_educated = krcw_tx } }
		if = { limit = { has_variable = college_educated } set_temp_variable = { krcw_tx = college_educated } multiply_temp_variable = { krcw_tx = krcw_tp } add_to_temp_variable = { krcw_ts_college_educated = krcw_tx } }
		if = {
			limit = { is_core_of = ROOT is_controlled_by = ROOT }
			if = {
				limit = { check_variable = { ROOT.krcw_dev_plan = 3 } }
				if = { limit = { has_variable = literate_educated } add_to_variable = { literate_educated = 0.05 } clamp_variable = { var = literate_educated min = 0 max = 100 } }
				if = { limit = { has_variable = primary_educated } add_to_variable = { primary_educated = 0.05 } clamp_variable = { var = primary_educated min = 0 max = 100 } }
			}
			if = {
				limit = { has_variable = krcw_dev_lvl check_variable = { krcw_dev_lvl > -1 } check_variable = { krcw_dev_lvl < 5 } }
				set_temp_variable = { krcw_tx = 5 }
				subtract_from_temp_variable = { krcw_tx = krcw_dev_lvl }
				multiply_temp_variable = { krcw_tx = krcw_tp }
				if = {
					limit = { check_variable = { krcw_tx > krcw_t_best } }
					set_temp_variable = { krcw_t_best = krcw_tx }
					ROOT = { set_variable = { krcw_dev_target = PREV } }
				}
			}
			if = {
				limit = { check_variable = { infrastructure_level < 4 } check_variable = { krcw_tp > krcw_t_rail } }
				set_temp_variable = { krcw_t_rail = krcw_tp }
				ROOT = { set_variable = { krcw_dev_rail_target = PREV } }
			}
		}
	}
	if = { limit = { check_variable = { krcw_t_pop < 0.01 } } set_temp_variable = { krcw_t_pop = 0.01 } }
	if = { limit = { has_variable = krcw_dev_pop } set_variable = { krcw_dev_pop_growth = krcw_t_pop } subtract_from_variable = { krcw_dev_pop_growth = krcw_dev_pop } multiply_variable = { krcw_dev_pop_growth = 1000 } }
	set_variable = { krcw_dev_pop = krcw_t_pop }
	set_variable = { krcw_dev_fac = num_of_civilian_factories }
	add_to_variable = { krcw_dev_fac = num_of_military_factories }
	add_to_variable = { krcw_dev_fac = num_of_naval_factories }
	set_variable = { krcw_dev_edu = 0 }
	set_temp_variable = { krcw_tx = krcw_ts_literate_educated }
	divide_temp_variable = { krcw_tx = krcw_dev_pop }
	set_variable = { krcw_dev_lit = krcw_tx }
	multiply_temp_variable = { krcw_tx = 0.45 }
	add_to_variable = { krcw_dev_edu = krcw_tx }
	set_temp_variable = { krcw_tx = krcw_ts_primary_educated }
	divide_temp_variable = { krcw_tx = krcw_dev_pop }
	set_variable = { krcw_dev_prim = krcw_tx }
	multiply_temp_variable = { krcw_tx = 0.1 }
	add_to_variable = { krcw_dev_edu = krcw_tx }
	set_temp_variable = { krcw_tx = krcw_ts_high_school_educated }
	divide_temp_variable = { krcw_tx = krcw_dev_pop }
	set_variable = { krcw_dev_hs = krcw_tx }
	multiply_temp_variable = { krcw_tx = 0.5 }
	add_to_variable = { krcw_dev_edu = krcw_tx }
	set_temp_variable = { krcw_tx = krcw_ts_college_educated }
	divide_temp_variable = { krcw_tx = krcw_dev_pop }
	set_variable = { krcw_dev_col = krcw_tx }
	multiply_temp_variable = { krcw_tx = 2.0 }
	add_to_variable = { krcw_dev_edu = krcw_tx }
	clamp_variable = { var = krcw_dev_edu min = 0 max = 100 }
}

# welfare moves toward 0.6 economy + 0.4 education; plans, war and hunger push it
krcw_dev_welfare = {
	set_temp_variable = { krcw_tx = krcw_econ }
	multiply_temp_variable = { krcw_tx = 0.6 }
	set_temp_variable = { krcw_ty = krcw_dev_edu }
	multiply_temp_variable = { krcw_ty = 0.4 }
	add_to_temp_variable = { krcw_tx = krcw_ty }
	subtract_from_temp_variable = { krcw_tx = krcw_dev_wel }
	multiply_temp_variable = { krcw_tx = 0.03 }
	set_variable = { krcw_dev_w_base = krcw_tx }
	set_variable = { krcw_dev_w_plan = 0 }
	if = { limit = { check_variable = { krcw_dev_plan = 1 } } set_variable = { krcw_dev_w_plan = -0.3 } }
	if = { limit = { check_variable = { krcw_dev_plan = 2 } } set_variable = { krcw_dev_w_plan = 0.4 } }
	if = { limit = { check_variable = { krcw_dev_plan = 4 } } set_variable = { krcw_dev_w_plan = -0.3 } }
	set_variable = { krcw_dev_w_war = 0 }
	if = { limit = { has_war = yes } set_variable = { krcw_dev_w_war = -0.4 } }
	if = { limit = { surrender_progress > 0.1 } set_variable = { krcw_dev_w_war = -1.0 } }
	set_variable = { krcw_dev_w_food = 0 }
	if = { limit = { check_variable = { resource@rubber < 0 } } set_variable = { krcw_dev_w_food = -0.5 } }
	set_variable = { krcw_dev_w_trend = krcw_dev_w_base }
	add_to_variable = { krcw_dev_w_trend = krcw_dev_w_plan }
	add_to_variable = { krcw_dev_w_trend = krcw_dev_w_war }
	add_to_variable = { krcw_dev_w_trend = krcw_dev_w_food }
	clamp_variable = { var = krcw_dev_w_trend min = -3 max = 3 }
	add_to_variable = { krcw_dev_wel = krcw_dev_w_trend }
	clamp_variable = { var = krcw_dev_wel min = 0 max = 100 }
}

			else_if = { limit = { check_variable = { krcw_econ_system = 2 } } krcw_econ_heat_down = yes krcw_econ_down_small = yes }
			else = {
				krcw_econ_up_small = yes
				if = { limit = { has_variable = krcw_rouble_debt } subtract_from_variable = { krcw_rouble_debt = 5 } clamp_variable = { var = krcw_rouble_debt min = 0 max = 100 } }
				add_stability = -0.02
			}
			set_variable = { krcw_dev_last = 7 }
			if = { limit = { is_ai = no } country_event = { id = krcw_econ.20 days = 1 } }
		}
	}
	if = {
		limit = { check_variable = { krcw_dev_p_special > 0 } }
		subtract_from_variable = { krcw_dev_p_special = 1 }
		if = {
			limit = { check_variable = { krcw_dev_p_special < 1 } }
			set_variable = { krcw_dev_p_special = 0 }
			subtract_from_variable = { krcw_dev_civ_used = krcw_dev_civ_special }
			set_variable = { krcw_dev_civ_special = 0 }
			subtract_from_variable = { krcw_dev_slots = 1 }
			if = {
				limit = { check_variable = { krcw_econ_system = 1 } }
				krcw_econ_up_small = yes
				random_country = {
					limit = { NOT = { tag = ROOT } krcw_econ_pcu_member = yes has_variable = krcw_econ }
					krcw_econ_up_small = yes
					add_opinion_modifier = { target = ROOT modifier = krcw_econ_aid_received }
				}
			}
			else_if = {
				limit = { check_variable = { krcw_econ_system = 2 } }
				krcw_econ_up_medium = yes
				add_to_variable = { krcw_econ_heat = 10 }
				clamp_variable = { var = krcw_econ_heat min = 0 max = 100 }
			}
			else = {
				krcw_econ_up_medium = yes
				add_to_variable = { krcw_dev_wel = -2 }
				if = { limit = { krcw_is_ma = yes } add_to_variable = { krcw_rouble_debt = 5 } }
			}
			set_variable = { krcw_dev_last = 8 }
			if = { limit = { is_ai = no } country_event = { id = krcw_econ.20 days = 1 } }
		}
	}
	if = { limit = { check_variable = { krcw_dev_t_uni > 0 } } subtract_from_variable = { krcw_dev_t_uni = 1 } }
	if = { limit = { check_variable = { krcw_dev_t_health > 0 } } subtract_from_variable = { krcw_dev_t_health = 1 } }
	if = { limit = { check_variable = { krcw_dev_t_goods > 0 } } subtract_from_variable = { krcw_dev_t_goods = 1 } }
	if = { limit = { check_variable = { krcw_dev_civ_used < 0 } } set_variable = { krcw_dev_civ_used = 0 } }
	if = { limit = { check_variable = { krcw_dev_slots < 0 } } set_variable = { krcw_dev_slots = 0 } }
}

# AI: every 3 months pick a national plan and one project by need (scripted-GUI AI weights are not used)
krcw_dev_ai = {
	add_to_variable = { krcw_dev_ai_clock = 1 }
	if = {
		limit = { check_variable = { krcw_dev_ai_clock > 2 } }
		set_variable = { krcw_dev_ai_clock = 0 }
		set_temp_variable = { krcw_t_plan = 0 }
		if = { limit = { OR = { has_war = yes AND = { has_variable = global.krcw_tension check_variable = { global.krcw_tension > 85 } } } } set_temp_variable = { krcw_t_plan = 4 } }
		else_if = { limit = { check_variable = { krcw_dev_wel < 35 } } set_temp_variable = { krcw_t_plan = 2 } }
		else_if = { limit = { check_variable = { krcw_dev_edu < 35 } } set_temp_variable = { krcw_t_plan = 3 } }
		else_if = { limit = { check_variable = { krcw_dev_wel > 55 } check_variable = { krcw_econ < 55 } } set_temp_variable = { krcw_t_plan = 1 } }
		if = { limit = { check_variable = { krcw_t_plan = 0 } krcw_dev_can_plan_0 = yes has_political_power > 100 } krcw_dev_set_plan_0 = yes }
		if = { limit = { check_variable = { krcw_t_plan = 1 } krcw_dev_can_plan_1 = yes has_political_power > 100 } krcw_dev_set_plan_1 = yes }
		if = { limit = { check_variable = { krcw_t_plan = 2 } krcw_dev_can_plan_2 = yes has_political_power > 100 } krcw_dev_set_plan_2 = yes }
		if = { limit = { check_variable = { krcw_t_plan = 3 } krcw_dev_can_plan_3 = yes has_political_power > 100 } krcw_dev_set_plan_3 = yes }
		if = { limit = { check_variable = { krcw_t_plan = 4 } krcw_dev_can_plan_4 = yes has_political_power > 100 } krcw_dev_set_plan_4 = yes }
		if = {
			limit = { has_political_power > 120 check_variable = { krcw_dev_slots < krcw_dev_slots_max } }
			if = { limit = { OR = { check_variable = { krcw_econ_bloat > 35 } check_variable = { krcw_econ_heat > 70 } } krcw_dev_can_reform = yes } krcw_dev_start_reform = yes }
			else_if = { limit = { check_variable = { krcw_dev_wel < 35 } krcw_dev_can_housing = yes } krcw_dev_start_housing = yes }
			else_if = { limit = { check_variable = { krcw_dev_edu < 50 } krcw_dev_can_literacy = yes } krcw_dev_start_literacy = yes }
			else_if = { limit = { always = yes krcw_dev_can_develop = yes } krcw_dev_start_develop = yes }
			else_if = { limit = { check_variable = { krcw_dev_wel < 55 } krcw_dev_can_health = yes } krcw_dev_start_health = yes }
			else_if = { limit = { check_variable = { krcw_econ < 45 } krcw_dev_can_goods = yes } krcw_dev_start_goods = yes }
			else_if = { limit = { always = yes krcw_dev_can_rail = yes } krcw_dev_start_rail = yes }
			else_if = { limit = { check_variable = { krcw_dev_edu > 60 } krcw_dev_can_university = yes } krcw_dev_start_university = yes }
		}
	}
}
