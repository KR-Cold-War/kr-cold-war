	}
	# Sacramento inherited a permanent seat from CWIC's United States by accident; it never signed a ceasefire
	if = { limit = { NOT = { has_global_flag = krcw_gv_psa_bonus } country_exists = PSA } set_global_flag = krcw_gv_psa_bonus PSA = { add_stability = 0.07 } }
}

krcw_gv_monthly = {
	if = { limit = { NOT = { has_global_flag = krcw_gv_status_done } } set_global_flag = krcw_gv_status_done krcw_gv_status_init = yes }
	# a save from before the regime, or a start-up that did not run: set it up on the first pulse
	if = { limit = { NOT = { has_global_flag = krcw_gv_active } NOT = { has_global_flag = krcw_gv_closed } } krcw_gv_startup = yes }
	if = {
		limit = { NOT = { has_global_flag = krcw_gv_closed } OR = { has_global_flag = KRCW_world_war has_global_flag = KRCW_cold_war_over } }
		set_global_flag = krcw_gv_closed
		clr_global_flag = krcw_gv_open
		if = {
			limit = { has_global_flag = KRCW_world_war }
			if = { limit = { check_variable = { global.krcw_gv_elbe_state < 2 } } krcw_gv_elbe_break = yes }
			if = { limit = { check_variable = { global.krcw_gv_ohio_state < 2 } } krcw_gv_ohio_break = yes }
			if = { limit = { check_variable = { global.krcw_gv_qd_state < 2 } } krcw_gv_qd_break = yes }
			random_country = { limit = { has_country_flag = KRCW_country } news_event = { id = krcw_gv.40 days = 1 } }
		}
		else = {
			set_global_flag = krcw_gv_final
			random_country = { limit = { has_country_flag = KRCW_country } news_event = { id = krcw_gv.41 days = 1 } }
		}
	}
	if = {
		limit = { has_global_flag = krcw_gv_active NOT = { has_global_flag = krcw_gv_closed } }
		krcw_gv_elbe_monthly = yes
		krcw_gv_ohio_monthly = yes
		krcw_gv_qd_monthly = yes
		if = {
			limit = { has_global_flag = krcw_gv_air_rules check_variable = { global.krcw_berlin_pressure > 20 } NOT = { has_global_flag = KRCW_berlin_war } }
			subtract_from_variable = { global.krcw_berlin_pressure = 1 }
		}
		if = { limit = { has_global_flag = krcw_gv_open } krcw_gv_close_session = yes }
		else = {
			add_to_variable = { global.krcw_gv_month = 1 }
			set_variable = { global.krcw_gv_wait = 9 }
			subtract_from_variable = { global.krcw_gv_wait = global.krcw_gv_month }
			if = { limit = { check_variable = { global.krcw_gv_month > 8 } } krcw_gv_open_session = yes }
		}
		krcw_gv_neutrals = yes
	}
}

krcw_gv_neutrals = {
	every_country = {
		limit = { has_idea = krcw_ceasefire_guarantor is_ai = yes }
		if = { limit = { krcw_gv_can_mediate = yes has_political_power > 90 } random_list = { 7 = { krcw_gv_mediate = yes } 93 = { } } }
		if = { limit = { krcw_gv_can_investigate = yes } random_list = { 12 = { krcw_gv_investigate = yes } 88 = { } } }
	}
	if = {
		limit = { NOT = { has_global_flag = krcw_gv_open } }
		random_country = {
			limit = { is_ai = yes krcw_is_ti = yes check_variable = { global.krcw_gv_prop_ti = 0 } has_political_power > 80 OR = { tag = FRA tag = ENG tag = WGR tag = ITA tag = CHI tag = WCA } }
			random_list = { 6 = { krcw_gv_ai_propose = yes } 94 = { } }
		}
		random_country = {
			limit = { is_ai = yes krcw_is_ma = yes check_variable = { global.krcw_gv_prop_ma = 0 } has_political_power > 80 OR = { tag = SOV tag = RAJ tag = USA tag = DDR } }
			random_list = { 6 = { krcw_gv_ai_propose = yes } 94 = { } }
		}
	}
}
